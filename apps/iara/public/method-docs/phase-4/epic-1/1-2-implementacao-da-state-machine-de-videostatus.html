Story 1.2: Implementacao da State Machine de VideoStatus

Status: done

Story

Como um sistema, Eu quero uma state machine que gerencie as transicoes de status dos videos, Para que todas as mudancas de estado sejam validadas e previsiveis.

Acceptance Criteria

AC1: Estados definidos corretamente

Given os estados do ciclo de vida do video When eu importo os tipos de src/lib/video-state-machine/types.ts Then tenho acesso ao tipo VideoStatus com valores:

  • 'new' - video recem-detectado
  • 'processing' - processamento LLM em andamento
  • 'draft' - metadados gerados, aguardando revisao
  • 'ready' - revisado e pronto para publicar
  • 'sending' - publicacao em andamento
  • 'sent' - publicado com sucesso (terminal)

AC2: Acoes de transicao definidas

Given as acoes possiveis no sistema When eu importo os tipos de src/lib/video-state-machine/types.ts Then tenho acesso ao tipo VideoAction com valores:

  • 'classify' - classificar tipo de video (self-loop em new)
  • 'process' - iniciar processamento LLM
  • 'complete' - processamento concluido com sucesso
  • 'error' - erro durante operacao
  • 'edit' - usuario editou metadados
  • 'finalize' - marcar como pronto
  • 'send' - iniciar publicacao
  • 'success' - publicacao concluida

AC3: Transicao new -> processing

Given um video com status new When eu chamo transition('new', 'process') Then retorna 'processing'

AC4: Transicao processing -> draft

Given um video com status processing When eu chamo transition('processing', 'complete') Then retorna 'draft'

AC5: Transicao processing -> new (erro)

Given um video com status processing When eu chamo transition('processing', 'error') Then retorna 'new' (permite reprocessar)

AC6: Transicao draft -> processing (reprocessar)

Given um video com status draft When eu chamo transition('draft', 'process') Then retorna 'processing'

AC7: Transicao draft -> ready

Given um video com status draft When eu chamo transition('draft', 'finalize') Then retorna 'ready'

AC8: Transicao ready -> draft (editar)

Given um video com status ready When eu chamo transition('ready', 'edit') Then retorna 'draft'

AC9: Transicao ready -> sending

Given um video com status ready When eu chamo transition('ready', 'send') Then retorna 'sending'

AC10: Transicao sending -> sent

Given um video com status sending When eu chamo transition('sending', 'success') Then retorna 'sent'

AC11: Transicao sending -> ready (erro)

Given um video com status sending When eu chamo transition('sending', 'error') Then retorna 'ready' (permite retentar)

AC12: Estado terminal sent

Given um video com status sent When eu chamo qualquer acao de transicao Then lanca InvalidTransitionError

AC13: Transicao invalida new -> send

Given um video com status new When eu chamo transition('new', 'send') Then lanca InvalidTransitionError (nao pode pular etapas)

AC14: InvalidTransitionError com contexto

Given uma transicao invalida When o erro e lancado Then InvalidTransitionError contem:

  • currentStatus: status atual
  • action: acao tentada
  • message: mensagem descritiva

AC15: Testes table-driven completos

Given o arquivo de testes transitions.test.ts When eu executo pnpm test Then testes cobrem todas as transicoes validas And testes cobrem transicoes invalidas para cada estado And usa test.each() para table-driven tests

AC16: Funcao pura sem side effects

Given a funcao transition() When chamada multiplas vezes com mesmos parametros Then sempre retorna o mesmo resultado (pura) And nao modifica nenhum estado externo

Implementacao de Referencia

// src/lib/video-state-machine/types.ts
export type VideoStatus = 'new' | 'processing' | 'draft' | 'ready' | 'sending' | 'sent'

export type VideoAction = 'process' | 'complete' | 'error' | 'edit' | 'finalize' | 'send' | 'success'

export class InvalidTransitionError extends Error {
  constructor(
    public readonly currentStatus: VideoStatus,
    public readonly action: VideoAction
  ) {
    super(`Invalid transition: cannot apply action '${action}' to status '${currentStatus}'`)
    this.name = 'InvalidTransitionError'
  }
}

// src/lib/video-state-machine/transitions.ts
import type { VideoStatus, VideoAction } from './types'
import { InvalidTransitionError } from './types'

const transitions: Record<VideoStatus, Partial<Record<VideoAction, VideoStatus>>> = {
  new: { process: 'processing' },
  processing: { complete: 'draft', error: 'new' },
  draft: { process: 'processing', finalize: 'ready', edit: 'draft' },
  ready: { send: 'sending', edit: 'draft' },
  sending: { success: 'sent', error: 'ready' },
  sent: {}, // estado terminal - nenhuma transicao permitida
}

export function transition(current: VideoStatus, action: VideoAction): VideoStatus {
  const next = transitions[current]?.[action]
  if (!next) {
    throw new InvalidTransitionError(current, action)
  }
  return next
}

// src/lib/video-state-machine/index.ts
export { transition } from './transitions'
export { InvalidTransitionError } from './types'
export type { VideoStatus, VideoAction } from './types'

Testes Table-Driven

// src/lib/video-state-machine/transitions.test.ts
import { describe, it, expect, test } from 'vitest'
import { transition, InvalidTransitionError } from './index'
import type { VideoStatus, VideoAction } from './index'

describe('video state machine', () => {
  describe('valid transitions', () => {
    test.each<[VideoStatus, VideoAction, VideoStatus]>([
      ['new', 'process', 'processing'],
      ['processing', 'complete', 'draft'],
      ['processing', 'error', 'new'],
      ['draft', 'process', 'processing'],
      ['draft', 'finalize', 'ready'],
      ['draft', 'edit', 'draft'],
      ['ready', 'send', 'sending'],
      ['ready', 'edit', 'draft'],
      ['sending', 'success', 'sent'],
      ['sending', 'error', 'ready'],
    ])('transition(%s, %s) returns %s', (from, action, expected) => {
      expect(transition(from, action)).toBe(expected)
    })
  })

  describe('invalid transitions', () => {
    test.each<[VideoStatus, VideoAction]>([
      ['new', 'complete'],
      ['new', 'error'],
      ['new', 'edit'],
      ['new', 'finalize'],
      ['new', 'send'],
      ['new', 'success'],
      ['processing', 'process'],
      ['processing', 'edit'],
      ['processing', 'finalize'],
      ['processing', 'send'],
      ['processing', 'success'],
      ['draft', 'complete'],
      ['draft', 'error'],
      ['draft', 'send'],
      ['draft', 'success'],
      ['ready', 'process'],
      ['ready', 'complete'],
      ['ready', 'error'],
      ['ready', 'finalize'],
      ['ready', 'success'],
      ['sending', 'process'],
      ['sending', 'complete'],
      ['sending', 'edit'],
      ['sending', 'finalize'],
      ['sending', 'send'],
      ['sent', 'process'],
      ['sent', 'complete'],
      ['sent', 'error'],
      ['sent', 'edit'],
      ['sent', 'finalize'],
      ['sent', 'send'],
      ['sent', 'success'],
    ])('transition(%s, %s) throws InvalidTransitionError', (from, action) => {
      expect(() => transition(from, action)).toThrow(InvalidTransitionError)
    })
  })

  describe('InvalidTransitionError', () => {
    it('contains currentStatus and action', () => {
      try {
        transition('new', 'send')
      } catch (error) {
        expect(error).toBeInstanceOf(InvalidTransitionError)
        if (error instanceof InvalidTransitionError) {
          expect(error.currentStatus).toBe('new')
          expect(error.action).toBe('send')
          expect(error.message).toContain('new')
          expect(error.message).toContain('send')
        }
      }
    })
  })

  describe('pure function property', () => {
    it('returns same result for same inputs', () => {
      const result1 = transition('new', 'process')
      const result2 = transition('new', 'process')
      expect(result1).toBe(result2)
    })
  })
})
